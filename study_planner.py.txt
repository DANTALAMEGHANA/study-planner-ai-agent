from datetime import datetime

def get_subjects():
    subjects = {}
    n = int(input("Enter number of subjects: "))
    for i in range(n):
        name = input(f"Enter subject {i+1} name: ")
        difficulty = int(input("Difficulty level (1-Easy, 2-Medium, 3-Hard): "))
        subjects[name] = difficulty
    return subjects

def generate_study_plan(subjects, exam_date, daily_hours):
    today = datetime.today()
    days_left = (exam_date - today).days

    if days_left <= 0:
        print("❌ Exam date must be in the future.")
        return

    print(f"\n📅 Days left for exam: {days_left}")
    print("\n📘 Smart Study Plan:\n")

    total_weight = sum(subjects.values())

    for subject, difficulty in subjects.items():
        subject_hours = (difficulty / total_weight) * (daily_hours * days_left)
        print(f"{subject}: {round(subject_hours, 1)} hours total")

def main():
    print("🧠 Smart Study Planner AI Agent\n")

    subjects = get_subjects()

    exam_date_input = input("\nEnter exam date (YYYY-MM-DD): ")
    exam_date = datetime.strptime(exam_date_input, "%Y-%m-%d")

    daily_hours = float(input("Enter daily study hours: "))

    generate_study_plan(subjects, exam_date, daily_hours)

if __name__ == "__main__":
    main()
